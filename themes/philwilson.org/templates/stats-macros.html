{% macro wordcount(pages) %} {%- set wordcount_array = pages | map(attribute='word_count') %} {%- set wordcount = 0 %} {%- for wordcount_item in wordcount_array %} {%- set_global wordcount = wordcount + wordcount_item %} {%- endfor %} {{- wordcount -}} {% endmacro wordcount %} {% macro years_of_blogging(pages) %} {%- if pages | length == 0 -%} 0 {%- else %} {%- set sorted_pages = pages | sort(attribute='date') -%} {%- set first_post = sorted_pages | first -%} {%- set last_post = sorted_pages | last -%} {%- set first_year = first_post.date | date(format="%Y") | int -%} {%- set last_year = last_post.date | date(format="%Y") | int -%} {{- (last_year - first_year + 1) }} {%- endif -%} {% endmacro years_of_blogging %} {% macro average_time_to_read(pages) %} {%- set reading_time_array = pages | map(attribute='reading_time') %} {%- set readingtime = 0 %} {%- for reading_time_item in reading_time_array %} {%- set_global readingtime = readingtime + reading_time_item %} {%- endfor %} {%- set number_of_pages = pages | length %} {{ readingtime / number_of_pages | round }} {% endmacro average_time_to_read %}
